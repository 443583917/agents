# Calculate first 10000 terms of series 1 - 1/3 + 1/5 - 1/7 + ...
# Multiply the total by 4

n_terms = 10000
total = 0.0

for i in range(n_terms):
    denominator = 2 * i + 1
    if i % 2 == 0:
        total += 1.0 / denominator
    else:
        total -= 1.0 / denominator

result = total * 4
print(f"Sum of first {n_terms} terms of 1 - 1/3 + 1/5 - ... = {total}")
print(f"Multiplied by 4: {result}")


OUTPUT:
